\begin{abstract}
\noindent This specification defines the mathematical foundations, econometric estimators, spatial stochastic processes, machine learning ensembles, uncertainty quantification protocols, and production integration architecture for \textbf{GLITTERS Valuation Engine v1.1.0} (\texttt{glitters\_valuation\_engine}), incorporating all **Phase 1: Evidence Integrity \& Statistical Correctness** audit remediations. Designed for Kenya's property markets across all 47 counties, the engine unifies seven valuation paradigms while enforcing five critical mathematical and engineering invariants: (1) strict segregation of capital sale evidence from rental cash-flow observations; (2) immutable SHA-256 model training provenance that blocks synthetic data from production training by default and propagates synthetic provenance into every inference output; (3) model-level multi-tenant scope binding (\texttt{PUBLIC\_REGISTRY\_ONLY}, \texttt{SINGLE\_TENANT\_PRIVATE}, \texttt{AUTHORIZED\_POOLED}) preventing cross-tenant weight leakage; (4) exact finite-sample order-statistic conformal calibration $k = \lceil (n_{\text{cal}} + 1)(1 - \alpha) \rceil$ requiring $n_{\text{cal}} \ge \lceil (1 - \alpha)/\alpha \rceil$ coupled with locally weighted KNN conformal scaling; and (5) global chronological sorting prior to locality MAD outlier screening and strictly causal historical reference filtering ($t_{\text{comp}} \le t_{\text{val}}$) with zero temporal lookahead leakage.
\end{abstract}

\takeaway{Statutory \& Non-Certification Compliance Invariant}{All outputs generated by \texttt{glitters\_valuation\_engine} are quantitative decision-support estimates and do \textbf{not} constitute a statutory valuation report certified by a Registered and Practising Valuer under the Valuers Act (Cap 532, Laws of Kenya) or the \href{https://isk.or.ke/membership/members-chapter/}{Institution of Surveyors of Kenya (ISK) Valuation Standards (2021)}. By default, \texttt{PropertyRecord.title\_status} initializes to \texttt{UNVERIFIED\_MISSING} and requires an explicit \texttt{TitleVerificationMetadata} audit record from official Ministry of Lands (Ardhisasa) searches before a title is treated as verified.}

\section{Core Valuation Problem \& Kenyan Market Formulation}
% ============================================================================

Let $\mathcal{P}$ denote the space of real property parcels and built units across Kenya's 47 counties. Each subject property $i \in \mathcal{P}$ is represented at valuation timestamp $t \in \mathbb{R}$ by the canonical feature tuple:
\begin{equation}
    \mathcal{X}_i(t) = \bigl(\mathbf{s}_i, c_i, \mathbf{x}_{\text{struct},i}, \mathbf{x}_{\text{loc},i}, \mathbf{x}_{\text{zone},i}, \mathbf{x}_{\text{tenure},i}, \mathbf{x}_{\text{inc},i}, \mathbf{x}_{\text{macro}}(t)\bigr),
\end{equation}
where:
\begin{itemize}
    \item $\mathbf{s}_i = (\phi_i, \lambda_i) \in [-4.75^\circ, 5.05^\circ] \times [33.90^\circ, 41.95^\circ]$ is the WGS84 latitude-longitude coordinate pair mapped to county, sub-county, and neighbourhood locality $L(i)$.
    \item $c_i \in \mathcal{C}$ is the property category across ten mandatory classes:
    \begin{multline*}
        \mathcal{C} = \{\text{Residential}, \text{Commercial}, \text{Industrial}, \text{Agricultural}, \text{Vacant Land},\\
        \text{Apartment/Condo}, \text{Rental}, \text{Development Land}, \text{Investment}, \text{Distressed}\}.
    \end{multline*}
    \item $\mathbf{x}_{\text{struct},i} = (A_{\text{land},i}, A_{\text{bldg},i}, B_{\text{bed},i}, B_{\text{bath},i}, \text{Age}_i, Q_{\text{cond},i}, Q_{\text{const},i})$ captures metric land area $A_{\text{land},i}$ ($\text{m}^2$), gross building floor area $A_{\text{bldg},i}$ ($\text{m}^2$), bedrooms, bathrooms, physical age $\text{Age}_i$ (years), condition score $Q_{\text{cond},i} \in [0,1]$, and construction quality $Q_{\text{const},i} \in [0,1]$.
    \item $\mathbf{x}_{\text{loc},i} = (I_{\text{access},i}, I_{\text{infra},i}, I_{\text{util},i}, \text{Tier}_i, d_{\text{CBD}}(\mathbf{s}_i))$ captures accessibility, paved road/drainage infrastructure, utilities (piped water, mains sewer, 3-phase electricity, fibre), neighbourhood tier $\text{Tier}_i \in \{1,2,3,4\}$ (aligned with \href{https://www.kba.co.ke/house-prices-drop-eases-homeownership-creates-challenges-for-developers/}{Kenya Bankers Association Housing Price Index (KBA-HPI) regional clustering}), and great-circle distance to the nearest urban economic hub.
    \item $\mathbf{x}_{\text{zone},i} = (Z_i, \text{GC}_{\max,i}, \text{FAR}_{\max,i}, H_{\max,i})$ defines permitted zoning category $Z_i$, maximum ground coverage ratio $\text{GC}_{\max,i} \in (0,1]$, maximum Plot Ratio / Floor Area Ratio $\text{FAR}_{\max,i} > 0$, and maximum storeys $H_{\max,i}$ under the \href{https://kara.or.ke/wp-content/uploads/2024/11/PhysicalandLandUsePlanningAct13of2019.pdf}{Physical and Land Use Planning Act, No. 13 of 2019 (PLUPA)} and county development control policies such as the \href{https://nairobiassembly.go.ke/ncca/wp-content/uploads/paperlaid/2023/SESSIONAL-PAPER-NO.-1-OF-2023-ON-NAIROBI-CITY-COUNTY-DEVELOPMENT-CONTROL-POLICY.pdf}{Nairobi City County Development Control Policy (Sessional Paper No. 1 of 2023)}.
    \item $\mathbf{x}_{\text{tenure},i} = (\tau_i, T_{\text{lease},i}, \Omega_{\text{title},i}, \mathcal{M}_{\text{title},i})$ specifies tenure type $\tau_i \in \{\text{Freehold}, \text{Leasehold}, \text{Sectional Title}, \text{Customary}, \text{Informal}\}$, unexpired leasehold term $T_{\text{lease},i}$ (years), title verification state $\Omega_{\text{title},i}$ (defaulting to \texttt{UNVERIFIED\_MISSING}), and optional audit metadata $\mathcal{M}_{\text{title},i}$ (\texttt{registry\_reference}, \texttt{verification\_source}, \texttt{verified\_timestamp\_iso}, \texttt{evidence\_document\_hash}).
\end{itemize}

\subsection{Strict Separation of Capital Sale Evidence vs.\ Rental Evidence}
The evidence space $\mathcal{E}$ is partitioned into two disjoint subsets:
\begin{align}
    \mathcal{E}_{\text{sale}} &= \{\texttt{VERIFIED\_STAMP\_DUTY\_SALE},\; \texttt{VERIFIED\_CONTRACT\_SALE},\; \texttt{VALUER\_HISTORICAL\_RECORD},\; \texttt{ASKING\_LISTING\_PRICE}\}, \\
    \mathcal{E}_{\text{rent}} &= \{\texttt{RENTAL\_LEASE\_CONTRACT},\; \texttt{RENTAL\_ASKING\_QUOTE}\}, \qquad \mathcal{E}_{\text{sale}} \cap \mathcal{E}_{\text{rent}} = \emptyset.
\end{align}
Observations with $E_j \in \mathcal{E}_{\text{sale}}$ enter the capital-value training and comparable-sales pipelines, with asking prices adjusted by configurable negotiation haircut $\kappa_{\text{ask}} \in \texttt{ValuationMarketConfig}$ (empirically estimable via \texttt{ValuationMarketConfig.calibrate\_from\_transactions()}). Observations with $E_j \in \mathcal{E}_{\text{rent}}$ carry an explicit \texttt{RentalPeriodUnit} $\in \{\texttt{MONTHLY}, \texttt{ANNUAL}, \texttt{PER\_SQM\_MONTHLY}\}$ and are routed exclusively to \texttt{IncomeValuationEngine.estimate\_market\_rent\_from\_comparables()}.

\subsection{Structural Leasehold Tenure Relativity}
We define the actuarial leasehold relativity factor $f_{\text{lease}}(\tau_i, T_{\text{lease},i}) \in (0, 1]$ relative to perpetual freehold at real discount rate $r_L$ (default $0.09$), with a statutory renewal friction deduction when $T_{\text{lease},i} < 25$ years:
\begin{equation}
    f_{\text{lease}}(\tau_i, T_{\text{lease},i}) =
    \begin{cases}
        1.00, & \text{if } \tau_i = \text{Freehold}, \\
        \bigl(1 - (1 + r_L)^{-T_{\text{lease},i}}\bigr) - 0.08 \max\!\Bigl(0, \dfrac{25 - T_{\text{lease},i}}{25}\Bigr), & \text{if } \tau_i \in \{\text{Leasehold}, \text{Sectional Title}\}.
    \end{cases}
\end{equation}
To guarantee strict economic monotonicity across all valuation paradigms (verified across 60 randomized trials in \texttt{test\_06\_sixty\_trial\_property\_based\_fuzzing}), $f_{\text{lease}}(\tau_i, T_{\text{lease},i})$ is applied structurally as a multiplicative relativity factor across Hedonic, Spatial GP, GBDT, Income, Cost, and Residual Land models rather than as an unconstrained collinear regression covariate.

\section{Mathematical Formulations of the Seven Valuation Approaches}
% ============================================================================

\subsection{Approach A: Semi-Log Regularized Hedonic Model with Duan's Smearing}
Fitted strictly on capital sales $\mathcal{D}_{\text{sale}}$, the hierarchical semi-log hedonic model with Empirical Bayes locality random intercepts $\alpha_{L(i)}$ and standardized structural/locational covariates $\tilde{\boldsymbol{\phi}}(\mathcal{X}_i) \in \mathbb{R}^{22}$ is:
\begin{equation}
    \ln V_i = \beta_0 + \alpha_{L(i)} + \tilde{\boldsymbol{\phi}}(\mathcal{X}_i)^\top \boldsymbol{\beta} + \varepsilon_i, \qquad \hat{V}_{\text{hedonic},i} = \exp\bigl(\hat{\beta}_0 + \hat{\alpha}_{L(i)} + \tilde{\boldsymbol{\phi}}(\mathcal{X}_i)^\top \hat{\boldsymbol{\beta}}\bigr) \cdot \hat{S}_{\text{Duan}} \cdot f_{\text{lease}}(\tau_i, T_i),
\end{equation}
where $\hat{S}_{\text{Duan}} = \bigl(\sum_{j=1}^N w_j \exp(\hat{e}_j)\bigr) / \sum_{j=1}^N w_j$ corrects Jensen's inequality retransformation bias under non-normal residuals.

\subsection{Approach B: Causal Comparable Sales Intelligence Engine}
For subject property $i$ valued at timestamp $t_0$, candidate $j$ is admitted if and only if $E_j \in \mathcal{E}_{\text{sale}}$ and $t_j \le t_0$ (strictly preventing future lookahead leakage). The composite kernel weight is:
\begin{equation}
    K(i, j) = K_{\text{spatial}}(d_{ij}) \cdot K_{\text{temp}}(t_0 - t_j) \cdot K_{\text{struct}}(\mathbf{z}_i, \mathbf{z}_j) \cdot w_{\text{rel}}(E_j) \cdot \exp\bigl(-\gamma \max(0, \text{GAR}_{ij} - 0.25)\bigr).
\end{equation}
From the top $K$ comparables with normalized weights $\tilde{w}_{ij}$, the engine computes $\hat{V}_{\text{comp},i} = \sum_{j=1}^K \tilde{w}_{ij} P_{j \to i}^{\text{adj}}$, Effective Sample Size $N_{\text{eff}} = (\sum_j K_{ij})^2 / \sum_j K_{ij}^2$, and Comparable Quality Score $\text{CQS}_i \in [0,1]$.

\subsection{Approach C: Scalable Inducing-Point Sparse Spatial Gaussian Process}
To eliminate the $\mathcal{O}(N^3)$ computational bottleneck and $\mathcal{O}(N^2)$ memory footprint of exact Kriging on large GLITTERS transaction histories, \texttt{SpatialGPValuationModel} implements the **Deterministic Training Conditional (DTC) / Nystr\"om Inducing-Point Sparse Gaussian Process** over $M \le 64$ spatially stratified landmark knots $\mathbf{Z} = \{\mathbf{u}_1, \dots, \mathbf{u}_M\} \subset \mathbb{R}^2$. Approximating the Mat\'ern($\nu=3/2$) covariance matrix as $\mathbf{K}_{NN} \approx \mathbf{K}_{NM} \mathbf{K}_{MM}^{-1} \mathbf{K}_{MN} + \sigma_n^2 \mathbf{I}_N$, the Woodbury matrix identity reduces inversion to the $M \times M$ system:
\begin{equation}
    \boldsymbol{\Sigma}_M = \mathbf{K}_{MM} + \frac{1}{\sigma_n^2} \mathbf{K}_{MN} \mathbf{K}_{NM} \in \mathbb{R}^{M \times M}, \qquad \boldsymbol{\alpha}_M = \boldsymbol{\Sigma}_M^{-1} \left(\frac{1}{\sigma_n^2} \mathbf{K}_{MN} \mathbf{e}_{\text{trend}}\right).
\end{equation}
At target location $\mathbf{s}_*$, the predictive mean and posterior variance are computed in $\mathcal{O}(M)$ time:
\begin{equation}
    \mu_{\text{GP}}(\mathbf{s}_*) = \mathbf{x}(\mathbf{s}_*)^\top \hat{\boldsymbol{\beta}}_{\text{trend}} + \mathbf{k}_{*M}^\top \boldsymbol{\alpha}_M, \qquad \sigma_{\text{GP}}^2(\mathbf{s}_*) = (\sigma_f^2 + \sigma_n^2) - \mathbf{k}_{*M}^\top \bigl(\mathbf{K}_{MM}^{-1} - \boldsymbol{\Sigma}_M^{-1}\bigr) \mathbf{k}_{M*}.
\end{equation}
In our empirical scalability benchmark ($N = 2,500$ transactions, $M = 64$ inducing points), offline fitting completes in **$66.86\text{ ms}$** and single-property online inference completes in **$0.289\text{ ms}$**.

\subsection{Approaches D, E, F \& G: GBDT/RF, Rental-Integrated Income DCF, Cost DRC \& PLUPA 2019 Residual HBU}
\begin{itemize}
    \item \textbf{Second-Order GBDT + Bootstrap RF (\texttt{machine\_learning})}: Fits $M_{\text{tree}} = 35$ Newton-step boosted trees on $\mathcal{D}_{\text{sale}}$ alongside $B = 12$ bootstrap Random Forest trees to quantify non-parametric epistemic tree variance.
    \item \textbf{Income Capitalization \& 10-Year DCF (\texttt{income\_valuation})}: Estimates market rent from nearby $\mathcal{E}_{\text{rent}}$ comparables ($t_j \le t_0$) when subject rent is absent, blends $75\%$ contractual rent with $25\%$ market rental comparables when both exist, and averages Direct Capitalization ($\text{NOI}_1 / R_0$) with a 10-year DCF model benchmarked to the Central Bank of Kenya (CBK) risk-free rate.
    \item \textbf{Depreciated Replacement Cost (\texttt{cost\_valuation})}: Combines urban-gravity baseline land value with JBC/IQSK reproduction cost new less Ross-Heidecke physical deterioration, functional obsolescence, and external obsolescence.
    \item \textbf{Residual Land \& HBU (\texttt{land\_valuation})}: Maximizes discounted residual land value across legally permissible uses under \href{https://kara.or.ke/wp-content/uploads/2024/11/PhysicalandLandUsePlanningAct13of2019.pdf}{PLUPA 2019} Ground Coverage and Plot Ratio limits.
\end{itemize}

\section{Exact Finite-Sample \& Locally Adaptive Conformal Calibration}
% ============================================================================

Let $\mathcal{D}_{\text{cal}} = \{(\mathcal{X}_j, V_j)\}_{j=1}^{n_{\text{cal}}}$ denote the chronologically held-out calibration cohort where $t_{\text{train}} \le t_{\text{cal},1} \le \dots \le t_{\text{cal},n_{\text{cal}}}$. Each calibration observation $j$ is evaluated using only historical comparables available at $t_{\text{cal},j}$, yielding normalized non-conformity score $s_j = |V_j - \hat{V}_{\text{ens},j}| / \sigma_{\text{total},j}$.

\begin{theorem}[Exact Finite-Sample Conformal Order Statistic \& Minimum Sample Condition]
For target confidence level $1 - \alpha \in (0, 1)$, exchangeability implies $\mathbb{P}\bigl(s_{n_{\text{cal}}+1} \le s_{(k)}\bigr) = \frac{k}{n_{\text{cal}}+1} \ge 1 - \alpha$ if and only if the order-statistic index $k$ satisfies:
\begin{equation}
    k = \bigl\lceil (n_{\text{cal}} + 1)(1 - \alpha) \bigr\rceil \le n_{\text{cal}} \quad \Longleftrightarrow \quad n_{\text{cal}} \ge \left\lceil \frac{1 - \alpha}{\alpha} \right\rceil.
\end{equation}
For $1 - \alpha = 0.90$ ($\alpha = 0.10$), the minimum valid calibration sample size is $n_{\text{cal},\min} = 9$. Whenever $n_{\text{cal}} < 9$, a $90\%$ finite-sample empirical order statistic does not exist; \texttt{UncertaintyAndConfidenceEngine} therefore sets \texttt{is\_conformal\_calibrated = False}, emits an explicit \texttt{UNCALIBRATED\_PREDICTION\_INTERVAL} flag, and applies a conservative fallback multiplier without claiming conformal calibration.
\end{theorem}

When $n_{\text{cal}} \ge n_{\text{cal},\min}$ and standardized calibration features $\{\mathbf{z}_j\}_{j=1}^{n_{\text{cal}}}$ are present, the engine computes a locally adaptive conformal multiplier $\hat{q}_{1-\alpha}(\mathbf{z}_*)$ around target $\mathbf{z}_*$ using Gaussian kernel weights $w_j(\mathbf{z}_*) = \exp\bigl(-\|\mathbf{z}_* - \mathbf{z}_j\|_2^2 / (2D h_z^2)\bigr)$:
\begin{equation}
    \hat{q}_{1-\alpha}(\mathbf{z}_*) = \max\Bigl(0.85\,s_{(k)},\; 0.70\,s_{(k)} + 0.30\,\hat{Q}_{1-\alpha}^{\text{local}}\bigl(\{(s_j, w_j(\mathbf{z}_*))\}\bigr)\Bigr).
\end{equation}

\takeaway{Impact of Fixing Calibration Leakage (CRITICAL 4 \& 5)}{After fixing the locality-grouping split bug and enforcing causal historical calibration ($t_{\text{comp}} \le t_{\text{cal}}$) with exact order-statistic conformal quantiles, the out-of-sample 90\% prediction interval coverage ($\text{PICP}_{90}$) on the Rolling Temporal Benchmark improved from $79.22\%$ (Calibration Error $10.78\%$ in v1.0.0) to **$89.61\%$ (Calibration Error $0.39\%$ against the $90.0\%$ target in v1.1.0)**, while $\text{R}^2$ improved from $0.9387$ to $0.9598$ and MAE dropped to KES 12.52M.}
